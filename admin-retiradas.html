import { neon } from "@neondatabase/serverless"; import { createHash, createHmac, randomBytes, timingSafeEqual } from "node:crypto"; export const config = { api: { bodyParser: false } }; const COOKIE_NAME = "usdtmz_admin_session"; const RATE_MZN_PER_USDT = 64; const MIN_MZN = 64; const MAX_MZN = 40000; function getDatabaseUrl() { return ( process.env.URL_DO_BANCO_DE_DADOS || process.env.POSTGRES_URL || process.env.DATABASE_URL || process.env.POSTGRES_URL_NON_POOLING || process.env.DATABASE_URL_UNPOOLED ); } function safeCompare(a, b) { const A = Buffer.from(String(a)); const B = Buffer.from(String(b)); if (A.length !== B.length) { return false; } return timingSafeEqual(A, B); } function getCookie(req, name) { const cookies = req.headers.cookie || ""; const item = cookies .split(";") .map(v => v.trim()) .find(v => v.startsWith(`${name}=`)); if (!item) { return null; } return item.substring(name.length + 1); } function verifyAdminSession(req) { const secret = process.env.ADMIN_SESSION_SECRET; if (!secret) { return false; } const token = getCookie(req, COOKIE_NAME); if (!token) { return false; } const parts = token.split("."); if (parts.length !== 2) { return false; } const [data, signature] = parts; const expected = createHmac("sha256", secret) .update(data) .digest("base64url"); if (!safeCompare(signature, expected)) { return false; } try { const payload = JSON.parse( Buffer.from(data, "base64url").toString("utf8") ); if (!payload.exp || Date.now() > payload.exp) { return false; } if (payload.id !== "admin") { return false; } return true; } catch { return false; } } async function readRawBody(req) { if (Buffer.isBuffer(req.body)) { return req.body.toString("utf8"); } if (typeof req.body === "string") { return req.body; } const chunks = []; for await (const chunk of req) { chunks.push( Buffer.isBuffer(chunk) ? chunk : Buffer.from(chunk) ); } return Buffer.concat(chunks).toString("utf8"); } function jsonBody(raw) { if (!raw) { return {}; } try { return JSON.parse(raw); } catch { return null; } } function normalizeMethod(value) { const method = String(value || "") .trim() .toUpperCase(); if (method === "MPESA") { return "MPESA"; } if ( method === "EMOLA" || method === "E-MOLA" || method === "E MOLA" ) { return "EMOLA"; } return null; } function normalizePhone(value) { const phone = String(value || "") .trim() .replace(/\s+/g, ""); if (!/^(84|85|86|87)\d{7}$/.test(phone)) { return null; } return phone; } function normalizeMzn(value) { if ( typeof value === "number" && Number.isInteger(value) ) { return value; } const text = String(value || "").trim(); if (!/^\d+$/.test(text)) { return null; } const amount = Number(text); if (!Number.isSafeInteger(amount)) { return null; } return amount; } function calculateUsdt(amountMzn) { return Number( (amountMzn / RATE_MZN_PER_USDT).toFixed(6) ); } function generateOrderId() { const random = randomBytes(8) .toString("hex") .toUpperCase(); return `ADM-${Date.now()}-${random}`; } function getPagarConfig() { const baseUrl = process.env.PAGAR_API_BASE_URL || "https://api.pagar.co.mz/api/v1"; const apiKey = process.env.PAGAR_API_KEY; const signingSecret = process.env.PAGAR_SIGNING_SECRET; if (!apiKey || !signingSecret) { throw new Error( "PAGAR_API_KEY ou PAGAR_SIGNING_SECRET não configurado." ); } return { baseUrl: baseUrl.replace(/\/+$/, ""), apiKey, signingSecret }; } async function readPagarResponse(response) { const text = await response.text(); let data; try { data = text ? JSON.parse(text) : {}; } catch { data = { message: text || "Resposta inválida da Pagar." }; } if (!response.ok) { const error = new Error( data.message || "Pedido rejeitado pela Pagar API." ); error.code = data.error; error.requestId = data.requestId; error.status = response.status; throw error; } return data; } async function pagarGet(path) { const { baseUrl, apiKey } = getPagarConfig(); const response = await fetch( `${baseUrl}${path}`, { method: "GET", headers: { Authorization: `Bearer ${apiKey}` } } ); return readPagarResponse(response); } async function pagarPost( path, body, idempotencyKey ) { const { baseUrl, apiKey, signingSecret } = getPagarConfig(); const timestamp = Date.now().toString(); const nonce = randomBytes(18).toString("base64url"); const rawBody = JSON.stringify(body); const bodyHash = createHash("sha256") .update(rawBody) .digest("hex"); const url = `${baseUrl}${path}`; const canonicalPath = new URL(url).pathname; const canonical = [ timestamp, nonce, "POST", canonicalPath, bodyHash ].join("\n"); const signature = createHmac( "sha256", signingSecret ) .update(canonical) .digest("hex"); const response = await fetch( url, { method: "POST", headers: { Authorization: `Bearer ${apiKey}`, "Content-Type": "application/json", "Idempotency-Key": idempotencyKey, "X-Pagar-Timestamp": timestamp, "X-Pagar-Nonce": nonce, "X-Pagar-Signature": `v1=${signature}` }, body: rawBody } ); return readPagarResponse(response); } function getPaymentFromResponse(data) { if ( data && data.payment ) { return data.payment; } return null; } function extractWebhookPayment(event) { if (!event || typeof event !== "object") { return null; } if ( event.payment && typeof event.payment === "object" ) { return event.payment; } if ( event.data && event.data.payment && typeof event.data.payment === "object" ) { return event.data.payment; } if ( event.data && typeof event.data === "object" ) { return event.data; } return null; } function getEventType(event) { return String( event?.type || event?.event || event?.name || "" ).trim(); } function getPaymentReference(payment) { return String( payment?.reference || "" ).trim(); } function getPaymentId(payment) { return String( payment?.id || "" ).trim(); } function getPaymentStatus(payment) { return String( payment?.status || "" ).trim().toUpperCase(); } async function markOrderFromPayment( sql, payment, eventId = null ) { const reference = getPaymentReference(payment); const paymentId = getPaymentId(payment); const status = getPaymentStatus(payment); if (!reference) { return { updated: false, reason: "reference_missing" }; } const amountMzn = payment?.amountMzn; const orderRows = await sql` SELECT id, order_id, amount, usdt_amount, rate, status, pagar_payment_id FROM orders WHERE order_id = ${reference} LIMIT 1 `; if (orderRows.length === 0) { return { updated: false, reason: "order_not_found" }; } const order = orderRows[0]; const orderAmount = Number(order.amount); if ( amountMzn !== undefined && Number(amountMzn) !== orderAmount ) { return { updated: false, reason: "amount_mismatch" }; } const validStatuses = new Set([ "PENDING", "PROCESSING", "PAID", "FAILED", "CANCELLED", "RECONCILIATION_REQUIRED" ]); if (!validStatuses.has(status)) { return { updated: false, reason: "unknown_status" }; } const newEventId = eventId || null; await sql` UPDATE orders SET status = ${status}, pagar_payment_id = COALESCE( ${paymentId || null}, pagar_payment_id ), pagar_event_id = COALESCE( ${newEventId}, pagar_event_id ), updated_at = NOW() WHERE order_id = ${reference} `; return { updated: true, order_id: order.order_id, status }; } async function handleWebhook( req, res ) { const webhookSecret = process.env.PAGAR_WEBHOOK_SECRET; if (!webhookSecret) { return res.status(500).json({ success: false, message: "PAGAR_WEBHOOK_SECRET não configurado." }); } const rawBody = await readRawBody(req); const eventId = req.headers["pagar-event-id"]; const signatureHeader = req.headers["pagar-signature"] || ""; if (!eventId) { return res.status(400).json({ success: false, message: "Pagar-Event-Id ausente." }); } const parts = {}; for ( const item of String( signatureHeader ).split(",") ) { const index = item.indexOf("="); if (index === -1) { continue; } const key = item.slice(0, index).trim(); const value = item.slice(index + 1).trim(); parts[key] = value; } const timestamp = parts.t; const received = parts.v1; if ( !/^\d+$/.test( String(timestamp || "") ) || !/^[a-f0-9]{64}$/.test( String(received || "") ) ) { return res.status(401).json({ success: false, message: "Assinatura do webhook inválida." }); } const timestampSeconds = Number(timestamp); if ( !Number.isFinite(timestampSeconds) || Math.abs( Date.now() / 1000 - timestampSeconds ) > 300 ) { return res.status(401).json({ success: false, message: "Webhook expirado." }); } const expected = createHmac( "sha256", webhookSecret ) .update( `${timestamp}.${rawBody}` ) .digest("hex"); if ( !safeCompare( received, expected ) ) { return res.status(401).json({ success: false, message: "Assinatura do webhook inválida." }); } const event = jsonBody(rawBody); if (!event) { return res.status(400).json({ success: false, message: "JSON do webhook inválido." }); } const eventType = getEventType(event); const sql = neon(getDatabaseUrl()); /* * Primeiro garantimos que o event ID * só pode ser processado uma vez. */ const inserted = await sql` INSERT INTO pagar_webhook_events ( event_id, event_type, payment_id, reference, payload, processed_at, created_at ) VALUES ( ${String(eventId)}, ${eventType || null}, ${ getPaymentId( extractWebhookPayment(event) ) || null }, ${ getPaymentReference( extractWebhookPayment(event) ) || null }, ${event}, NOW(), NOW() ) ON CONFLICT (event_id) DO NOTHING RETURNING id `; /* * Webhook repetido: * já foi processado. */ if (inserted.length === 0) { return res.status(200).json({ success: true, duplicate: true }); } const payment = extractWebhookPayment(event); if (!payment) { return res.status(200).json({ success: true, processed: true, message: "Evento recebido sem pagamento associado." }); } const result = await markOrderFromPayment( sql, payment, String(eventId) ); return res.status(200).json({ success: true, processed: true, result }); } async function handleCreate( req, res, body ) { if (!verifyAdminSession(req)) { return res.status(401).json({ success: false, authenticated: false, message: "Sessão de administrador inválida ou expirada." }); } const amountMzn = normalizeMzn( body.amount_mzn ); const method = normalizeMethod( body.payment_method ); const payerPhone = normalizePhone( body.payerPhone ); if ( amountMzn === null ) { return res.status(400).json({ success: false, message: "O valor deve ser um número inteiro em MZN." }); } if ( amountMzn < MIN_MZN ) { return res.status(400).json({ success: false, message: `O valor mínimo é ${MIN_MZN} MZN.` }); } if ( amountMzn > MAX_MZN ) { return res.status(400).json({ success: false, message: `O valor máximo é ${MAX_MZN} MZN.` }); } if (!method) { return res.status(400).json({ success: false, message: "Método de pagamento inválido." }); } if (!payerPhone) { return res.status(400).json({ success: false, message: "Número M-Pesa/e-Mola inválido." }); } const databaseUrl = getDatabaseUrl(); if (!databaseUrl) { return res.status(500).json({ success: false, message: "Base de dados não configurada." }); } const sql = neon(databaseUrl); const orderId = generateOrderId(); const usdtAmount = calculateUsdt(amountMzn); const title = "Compra de USDTMZ"; const description = `Compra administrativa de ${usdtAmount} USDT ` + `à taxa de ${RATE_MZN_PER_USDT} MZN por USDT.`; /* * A referência, preço, título e quantidade * são criados pelo backend. */ await sql` INSERT INTO orders ( order_id, name, phone, operation, payment, amount, usdt_amount, rate, status, created_at, updated_at ) VALUES ( ${orderId}, 'USDTMZ ADMIN', ${payerPhone}, 'BUY_USDT_ADMIN', ${method}, ${amountMzn}, ${usdtAmount}, ${RATE_MZN_PER_USDT}, 'PENDING', NOW(), NOW() ) `; const pagarBody = { reference: orderId, title, description, amountMzn, method, payerPhone }; let pagarResult; try { pagarResult = await pagarPost( "/payments", pagarBody, `payment:${orderId}` ); } catch (error) { await sql` UPDATE orders SET status = 'FAILED', updated_at = NOW() WHERE order_id = ${orderId} `; return res.status( error.status || 502 ).json({ success: false, message: error.message || "Não foi possível criar o pagamento na Pagar.", code: error.code || null, requestId: error.requestId || null, order_id: orderId }); } const payment = getPaymentFromResponse( pagarResult ); if (!payment) { return res.status(502).json({ success: false, message: "A Pagar não devolveu o objeto payment.", order_id: orderId }); } const paymentId = getPaymentId(payment); const paymentStatus = getPaymentStatus(payment) || "PENDING"; await sql` UPDATE orders SET pagar_payment_id = ${paymentId || null}, status = ${paymentStatus}, updated_at = NOW() WHERE order_id = ${orderId} `; return res.status(202).json({ success: true, order_id: orderId, payment_id: paymentId || null, status: paymentStatus, amount_mzn: amountMzn, usdt_amount: usdtAmount, rate: RATE_MZN_PER_USDT, method, message: "Pagamento criado. Aguarde a confirmação PAID." }); } async function handleCheckStatus( req, res, body ) { if (!verifyAdminSession(req)) { return res.status(401).json({ success: false, authenticated: false, message: "Sessão de administrador inválida ou expirada." }); } const orderId = String( body.order_id || "" ).trim(); if (!orderId) { return res.status(400).json({ success: false, message: "order_id é obrigatório." }); } const sql = neon(getDatabaseUrl()); const rows = await sql` SELECT id, order_id, amount, usdt_amount, rate, status, payment, phone, pagar_payment_id, pagar_event_id, blockchain_tx_hash, wallet_address, updated_at FROM orders WHERE order_id = ${orderId} LIMIT 1 `; if (rows.length === 0) { return res.status(404).json({ success: false, message: "Pedido não encontrado." }); } const order = rows[0]; /* * Se já estiver PAID, não precisamos * fazer outra chamada à Pagar. */ if ( String(order.status) .toUpperCase() === "PAID" ) { return res.status(200).json({ success: true, order }); } let pagarData; try { if (order.pagar_payment_id) { pagarData = await pagarGet( `/payments/${encodeURIComponent( order.pagar_payment_id )}` ); } else { pagarData = await pagarGet( `/payments/by-reference/${encodeURIComponent( order.order_id )}` ); } } catch (error) { return res.status( error.status || 502 ).json({ success: false, message: error.message || "Não foi possível consultar o pagamento.", code: error.code || null, requestId: error.requestId || null }); } const payment = getPaymentFromResponse( pagarData ); if (!payment) { return res.status(502).json({ success: false, message: "A Pagar não devolveu o pagamento." }); } const reference = getPaymentReference(payment); if ( reference && reference !== order.order_id ) { return res.status(409).json({ success: false, message: "A referência devolvida pela Pagar não corresponde ao pedido." }); } const paymentAmount = Number(payment.amountMzn); if ( Number.isFinite(paymentAmount) && paymentAmount !== Number(order.amount) ) { return res.status(409).json({ success: false, message: "O valor confirmado pela Pagar não corresponde ao pedido." }); } const paymentId = getPaymentId(payment); const status = getPaymentStatus(payment); await sql` UPDATE orders SET pagar_payment_id = COALESCE( ${paymentId || null}, pagar_payment_id ), status = ${status}, updated_at = NOW() WHERE order_id = ${order.order_id} `; const updated = await sql` SELECT order_id, amount, usdt_amount, rate, status, payment, phone, pagar_payment_id, pagar_event_id, blockchain_tx_hash, wallet_address, updated_at FROM orders WHERE order_id = ${order.order_id} LIMIT 1 `; return res.status(200).json({ success: true, order: updated[0] }); } export default async function handler( req, res ) { if (req.method !== "POST") { return res.status(405).json({ success: false, message: "Método não permitido." }); } try { const raw = await readRawBody(req); const body = jsonBody(raw); if (!body) { return res.status(400).json({ success: false, message: "JSON inválido." }); } /* * O webhook é identificado pelo header * oficial da Pagar. */ const isWebhook = Boolean( req.headers["pagar-event-id"] || req.headers["pagar-signature"] ); if (isWebhook) { return await handleWebhook( req, res ); } if (body.check_status === true) { return await handleCheckStatus( req, res, body ); } return await handleCreate( req, res, body ); } catch (error) { console.error( "CRIAR COMPRA ERROR:", error ); return res.status(500).json({ success: false, message: "Erro interno ao processar a compra.", error: error?.message || "Erro desconhecido." }); } }
